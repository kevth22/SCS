rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /mitglieder/{username} {
      allow read: if true;
      allow create: if request.resource.data.rolle == "gast"
                    && request.resource.data.aktiv == true
                    && request.resource.data.benutzername == username;
      allow update: if true;
      allow delete: if false;
    }
    match /{document=**} {
      allow read, write: if true;
    }
  }
}

WICHTIG: Diese Regeln sind nur die aktuelle Übergangslösung des Kontosystems
ohne Firebase Authentication. In Firebase unter Firestore Database > Regeln
einfügen und veröffentlichen, damit Red Dart beim ersten Login automatisch
angelegt werden kann.
